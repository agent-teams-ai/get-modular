import datetime
import json
import os
import pathlib
import subprocess
import sys
import time

size = int(sys.argv[1])
assert size in (8, 100, 500, 1000)
root = pathlib.Path('/tmp/gm-ts-owner-20261020')
clone = pathlib.Path('/tmp/gm-patched-consumer-20261020')
results = root / 'consumer-results'
results.mkdir(exist_ok=True)
prefix = results / ('n' + str(size))
assert not prefix.with_suffix('.json').exists(), 'One recorded sample per selected size'
node = '/tmp/gm-completion-tools-20261020/node-v26.11.1-linux-x64/bin/node'
script = ('import {measureGrowth} from ' + json.dumps((clone / 'tests/qualification/consumer-fixtures/growth/run.mjs').as_uri()) + ';'
          + 'const r=await measureGrowth(' + json.dumps({'size':size,'compilerCandidate':str(clone / 'private-candidate.json'),'directoryParent':str(results)}) + ');'
          + 'console.log(JSON.stringify({event:"private.compiler-owner.complete",size:r.size,status:r.status,directory:r.directory}));')
command = [node, '--input-type=module', '--eval', script]
env = os.environ.copy()
env['PATH'] = '/tmp/gm-completion-tools-20261020/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin'
env['NODE_USE_ENV_PROXY'] = '1'
cg = pathlib.Path('/sys/fs/cgroup')
def memory():
    return {'currentBytes':int((cg / 'memory.current').read_text()),'max':(cg / 'memory.max').read_text().strip(),'events':(cg / 'memory.events').read_text()}

receipt = {'kind':'private-consumer-observation','size':size,'command':command,'cwd':str(clone),
           'startUtc':datetime.datetime.now(datetime.timezone.utc).isoformat(), 'memoryBefore':memory(),
           'rssMeaning':'Linux wait4 rusage.ru_maxrss: highest RSS of a waited process in this child lineage, including waited descendants; not simultaneous aggregate tree memory.',
           'compilerAndRuntimeLimits':'Existing run.mjs enforces compiler180s/runtime120s/aggregate600s unchanged.',
           'observerVersion':2}
start = time.monotonic()
with prefix.with_suffix('.stdout.log').open('w') as out, prefix.with_suffix('.stderr.log').open('w') as err:
    process = subprocess.Popen(command,cwd=clone,env=env,stdout=out,stderr=err,start_new_session=True)
    waited,status,usage = os.wait4(process.pid,0)
    assert waited == process.pid
    process.returncode = os.waitstatus_to_exitcode(status)
    receipt['rusagePeakRssBytes'] = usage.ru_maxrss * 1024
receipt.update({'exitCode':process.returncode,'wallSeconds':time.monotonic()-start,'endUtc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'memoryAfter':memory()})
for line in prefix.with_suffix('.stdout.log').read_text().splitlines():
    try: item=json.loads(line)
    except json.JSONDecodeError: continue
    if item.get('event')=='private.compiler-owner.complete': receipt['completion']=item
prefix.with_suffix('.json').write_text(json.dumps(receipt,indent=2)+'\n')
print(json.dumps(receipt),flush=True)
raise SystemExit(process.returncode)
