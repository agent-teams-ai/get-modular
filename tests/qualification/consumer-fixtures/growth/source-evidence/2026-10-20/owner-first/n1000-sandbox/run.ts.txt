import assert from "node:assert/strict";
import { readFile, writeFile } from "node:fs/promises";
import { checkGrowthSource } from "./source.ts";
const result = await checkGrowthSource(process.argv[2]!, 1000);
const retained = JSON.parse(await readFile(process.argv[3]!, "utf8")) as { readonly stages: readonly { readonly stage: string; readonly source: unknown }[] };
assert.deepEqual(result.sourceTree, retained.stages.find(item => item.stage === "cold")!.source);
await writeFile(process.argv[4]!, JSON.stringify(result, null, 2) + "\n");
console.log(JSON.stringify({ kind: result.kind, size: result.size, source: result.source, sourceTree: result.sourceTree,
  factCoverage: result.observation.source.factCoverage, completeInventory: result.observation.completeInventory,
  resolvedImports: result.observation.source.graph.edges.length, externalImports: result.observation.source.graph.external.length,
  features: result.observation.source.fms!.features.length, cms: result.observation.source.cms!.outcome,
  policy: result.observation.policy.report.outcome, cliMs: result.observation.policy.wallMs,
  measuredChainMs: result.observation.wallMs, topology: result.observation.topology.outcome }));
