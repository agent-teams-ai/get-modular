import type { Assembly } from "@get-modular/assembly";
import { assemblyFor } from "@get-modular/assembly";
import { scoped } from "@get-modular/resources";

type Capabilities = Record<never, never>;
declare const journalInput: import("@get-modular/core").ModuleDeclaration;
declare const sessionInput: import("@get-modular/core").ModuleDeclaration;
declare const storeDeclaration: import("@get-modular/core").ModuleDeclaration;
declare const cacheDeclaration: import("@get-modular/core").ModuleDeclaration;
declare const createStore: (...args: any[]) => any;
declare const createCache: (...args: any[]) => any;

export function compose(api: Assembly<Capabilities>) {
  const journal = api.bindInput(journalInput);
  const session = api.bindInput(sessionInput);
  const store = api.bindFactory(storeDeclaration, scoped(storeDeclaration.implementationId, createStore));
  const cache = api.bindFactory(cacheDeclaration, createCache);
  return api.prepare({
    profileId: "test/sessions", factories: [store, cache], roots: { cache },
    inputs: { journal, session },
  });
}

export const prepared = compose(assemblyFor<Capabilities>());
