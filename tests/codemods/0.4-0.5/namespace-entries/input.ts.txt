import * as gm from "@get-modular/assembly";
import * as resources from "@get-modular/resources";

type Capabilities = Record<never, never>;
declare const moduleDeclaration: import("@get-modular/core").ModuleDeclaration;
declare const inputDeclaration: import("@get-modular/core").ModuleDeclaration;
declare const factory: (...args: any[]) => any;

const api = gm.assemblyFor<Capabilities>();
export const scopedEntry = api.bindFactory(moduleDeclaration, resources.scoped(moduleDeclaration.implementationId, factory));
export const plainEntry = api.bindFactory(moduleDeclaration, factory);
export const supplied = api.bindInput(inputDeclaration);
