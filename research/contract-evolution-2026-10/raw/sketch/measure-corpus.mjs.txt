const schema = JSON.parse(readFileSync(new URL('../../../architecture/contracts/v1/composition.schema.json', import.meta.url)));
const pid = schema.$defs.portableId; const re = new RegExp(pid.pattern, 'u');
const isPortableId = v => typeof v === 'string' && v.length >= pid.minLength && v.length <= pid.maxLength && re.test(v);
const bytes = s => Buffer.byteLength(s, 'utf8');
const isRecord = v => v !== null && typeof v === 'object' && !Array.isArray(v);

function entriesOf(decl) {
  const out = [];
  if (!isRecord(decl)) return out;
  for (const side of ['provides', 'slots']) if (Array.isArray(decl[side])) for (const e of decl[side]) if (isRecord(e)) out.push({ side, e });
  return out;
}
function allocate(decls) {
  const per = new Map();
  for (const d of decls) for (const { e } of entriesOf(d)) {
    const c = e.compatibility; if (!isRecord(c) || !isPortableId(e.capabilityId) || !isPortableId(c.token)) continue;
    if (!per.has(e.capabilityId)) per.set(e.capabilityId, []);
    const list = per.get(e.capabilityId); if (!list.includes(c.token)) list.push(c.token);
  }
  const map = new Map();
  for (const [cap, tokens] of per) {
    const taken = new Set(); const m = new Map();
    for (const t of tokens) {
      let n; const b = t.match(/^(.*)\/r([1-9][0-9]*)$/u); const v = t.match(/\/v([1-9][0-9]*)$/u);
      if (b && b[1] === cap) n = Number(b[2]); else if (v) n = Number(v[1]); else if (t === cap) n = 1;
      if (n !== undefined && !taken.has(n)) { m.set(t, n); taken.add(n); }
    }
    let next = 1;
    for (const t of tokens) if (!m.has(t)) { while (taken.has(next)) next++; m.set(t, next); taken.add(next); }
    map.set(cap, m);
  }
  return map;
}
function versionClass(d) {
  if (!isRecord(d)) return 'non-record';
  if (!Object.hasOwn(d, 'schemaVersion')) return 'missing';
  const v = d.schemaVersion; if (v === 1) return '1'; if (v === 2) return '2';
  return Number.isInteger(v) ? 'other-int' : 'not-admitted';
}
function features(expected) {
  const diags = expected?.diagnostics ?? [];
  const s = JSON.stringify(diags);
  return {
    ok: expected?.ok === true, planBindings: expected?.plan?.bindings?.length ?? 0,
    mismatch: diags.filter(d => d.code === 'binding.compatibility-mismatch').length,
    compatPath: diags.filter(d => /"(compatibility|token|family|familyVersion)"/u.test(JSON.stringify(d.path))).length,
    limit: diags.filter(d => d.code === 'input.limit-exceeded').map(d => d.details?.limitName),
    unsupported: diags.filter(d => d.code === 'schema.unsupported-version').length,
  };
}
const BYTE_LIMITS = new Set(['declarationRawDocumentBytes', 'profileRawDocumentBytes', 'aggregateRawBytes', 'aggregateStringBytes', 'identifierBytes']);
function analyzeObject(row) {
  const decls = Array.isArray(row.input?.declarations) ? row.input.declarations : [];
  const alloc = allocate(decls);
  let eligible = 0, ineligible = 0, deltaA = 0, deltaB = 0, maxTok = 0;
  const versions = new Set();
  for (const d of decls) {
    versions.add(versionClass(d));
    for (const { side, e } of entriesOf(d)) {
      const c = e.compatibility;
      if (!isRecord(c)) continue;
      if (!isPortableId(e.capabilityId) || !isPortableId(c.token)) { ineligible++; continue; }
      eligible++;
      const n = alloc.get(e.capabilityId).get(c.token);
      const nt = `${e.capabilityId}/r${n}`; maxTok = Math.max(maxTok, bytes(nt));
      deltaA += bytes(nt) - bytes(c.token);
      const before = bytes(JSON.stringify({ compatibility: c })) - 2;
      const after = bytes(JSON.stringify(side === 'provides' ? { revision: n, compatibleFrom: n } : { revision: n })) - 2;
      deltaB += after - before;
    }
  }
  const f = features(row.expected);
  const withEntries = decls.some(d => entriesOf(d).length > 0);
  const nonAdmittedWithEntries = decls.some(d => ['missing', 'not-admitted'].includes(versionClass(d)) && entriesOf(d).length > 0);
  const v2 = decls.some(d => versionClass(d) === '2');
  const byteLimit = f.limit.some(l => BYTE_LIMITS.has(l));
  // Option A classification
  let A = 'unchanged';
  if (v2) A = f.unsupported > 0 ? 'rewrite-2-to-3' : 'retire:schemaVersion2';
  else if (maxTok > 128) A = 'retire:token>128';
  else if (f.limit.length && deltaA !== 0) A = 'retire:token-bytes-limit';
  else if (nonAdmittedWithEntries && f.compatPath > 0) A = 'retire:version-not-admitted';
  else if (eligible > 0 && (f.planBindings > 0 || f.mismatch > 0)) A = 'transform:plan-or-mismatch';
  else if (eligible > 0) A = 'transform:tokens-only';
  else if (f.ok) A = 'transform:plan-schema-only';
  // Option B classification
  let B = 'unchanged';
  if (v2) B = f.unsupported > 0 ? 'rewrite-2-to-3' : 'retire:schemaVersion2';
  else if (f.compatPath > 0) B = 'retire:compat-path';
  else if (f.limit.length && deltaB !== 0) B = 'retire:bytes-limit';
  else if (eligible + ineligible > 0) B = 'lift:' + (f.ok || f.mismatch ? 'plan-or-mismatch' : 'diagnostics-equal-up-to-shape');
  else if (f.ok) B = 'transform:plan-schema-only';
  return { A, B, eligible, ineligible, deltaA, deltaB, withEntries, compatPath: f.compatPath };
}
function tally(name, rows) {
  const a = {}, b = {}; let n = 0, elig = 0, cp = 0;
  for (const r of rows) { n++; const x = analyzeObject(r); a[x.A] = (a[x.A] ?? 0) + 1; b[x.B] = (b[x.B] ?? 0) + 1; if (x.eligible) elig++; if (x.compatPath) cp++; }
  console.log(JSON.stringify({ category: name, rows: n, rowsWithEligibleTokens: elig, rowsWithCompatPaths: cp, A: a, B: b }));
}

// Raw documents: parse utf8 recipes where possible.
function rawAnalyze(recipes, expected, caseId) {
  const decls = []; let unparsed = 0, withCompatText = 0, docsVersion = [];
  for (const r of recipes) {
    if (r?.kind !== 'utf8') { continue; }
    if (r.source.includes('"compatibility"')) withCompatText++;
    try { const v = JSON.parse(r.source); decls.push(v); docsVersion.push(versionClass(v)); } catch { unparsed++; decls.push(null); }
  }
  return { decls, unparsed, withCompatText, docsVersion };
}

  const a = {}, b = {}; let n = 0, compatDocs = 0;
  const detail = [];
  for (const row of rawDocumentCases()) {
    n++;
    const { decls, withCompatText } = rawAnalyze(row.inputRecipe.declarations, row.expected, row.caseId);
    const x = analyzeObject({ input: { declarations: decls.filter(Boolean) }, expected: row.expected });
    // raw-specific: lexeme lanes on schemaVersion
    let A = x.A, B = x.B;
    const src = row.inputRecipe.declarations.filter(r => r?.kind === 'utf8').map(r => r.source);
    const versionLexeme = src.map(s => (s.match(/"schemaVersion":([^,}]*)/u) ?? [])[1]).filter(Boolean);
    const odd = versionLexeme.filter(l => l !== '1');
    if (odd.length) {
      const projected2 = odd.some(l => { try { return JSON.parse(l) === 2; } catch { return false; } });
      const unsupported = (row.expected.diagnostics ?? []).some(d => d.code === 'schema.unsupported-version');
      const nonAdmitted = odd.some(l => { try { const v = JSON.parse(l); return !Number.isSafeInteger(v) || Object.is(v, -0); } catch { return true; } });
      if (projected2 && unsupported && odd.every(l => l === '2')) A = 'rewrite-2-to-3';
      else if (projected2) A = 'retire:schemaVersion2(lexeme)';

    } else if (withCompatText) B = 'lift:raw-text-edit';

    a[A] = (a[A] ?? 0) + 1; b[B] = (b[B] ?? 0) + 1;
