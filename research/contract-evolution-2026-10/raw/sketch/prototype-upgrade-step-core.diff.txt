+// Upgrade step for declaration schema 1 (ADR-NNNN). It presents a schema 1 document as schema 2 through an
+// overlay over the supplied document; admission then validates the one current shape. Every path stays the
+// path of the supplied document: synthetic members are valid by construction and never produce diagnostics.

+type Report = (violation: DocumentShapeViolation) => void;
+type Path = readonly (string | number)[];
+const synthetic: unique symbol = Symbol("previous-declaration-synthetic");
+const overlay: unique symbol = Symbol("previous-declaration-overlay");
+type Synthetic = { readonly [synthetic]: number };
+type RecordOverlay<Value> = {
+  readonly [overlay]: "record";
+  readonly base: Value;
+  readonly keys: readonly string[];
+  readonly members: ReadonlyMap<string, Upgraded<Value> | null>;
+};
+type ArrayOverlay<Value> = {
+  readonly [overlay]: "array";
+  readonly base: Value;
+  readonly items: ReadonlyMap<number, Upgraded<Value>>;

+export type Upgraded<Value> = Value | Synthetic | RecordOverlay<Value> | ArrayOverlay<Value>;

+function isSynthetic(value: unknown): value is Synthetic {
+  return typeof value === "object" && value !== null && Object.hasOwn(value, synthetic);

+function overlayKind(value: unknown): "record" | "array" | null {
+  return typeof value === "object" && value !== null && Object.hasOwn(value, overlay)
+    ? (value as { readonly [overlay]: "record" | "array" })[overlay] : null;

+/** The builder encoding `<capabilityId>/r<N>` with a canonical decimal N, or null. */
+export function builderRevision(capabilityId: string, token: string): number | null {
+  const prefix = `${capabilityId}/r`;
+  if (!token.startsWith(prefix)) {return null;}
+  const digits = token.slice(prefix.length);
+  if (!/^[1-9][0-9]{0,9}$/u.test(digits)) {return null;}
+  const value = Number(digits);
+  return value <= maximumRevision ? value : null;

+function upgradedReader<Value>(base: DocumentReader<Value>): DocumentReader<Upgraded<Value>> {
+  const asValue = (value: Upgraded<Value>): Value => value as Value;
+  const reader: DocumentReader<Upgraded<Value>> = {
+    kind: value => isSynthetic(value) ? "number" : overlayKind(value) ?? base.kind(asValue(value)),
+    keys: value => overlayKind(value) === "record" ? (value as RecordOverlay<Value>).keys : base.keys(asValue(value)),
+    own: (value, key): OwnMember<Upgraded<Value>> => {

+      }

+    item: (value, index) => {
