declare const brand: unique symbol;
type Key<V> = keyof V & string;
type Keys<V> = readonly [Key<V>, ...Key<V>[]];
type Snap<V, K extends keyof V> = { readonly revision: number; readonly values: Readonly<Pick<V, K>>; readonly [brand]: true };
type Snap3<V, K extends keyof V> = { readonly revision: number; readonly values: { readonly [P in K]: V[P] }; readonly [brand]: true };
type App = { readonly theme: string; readonly region: string };
type Wrong = { readonly theme: number };

type R1<V> = { readonly read: <const K extends Keys<V>>(keys: K) => Snap<V, K[number]> };
type R2<V> = { readonly read: <const K extends Key<V>>(keys: readonly [K, ...K[]]) => Snap<V, K> };
type R3<V> = { readonly read: <const K extends Key<V>>(keys: readonly [K, ...K[]]) => Snap3<V, K> };
type R6<V> = { readonly read: <const K extends Keys<V>>(keys: K) => Snap3<V, K[number]> };
declare const r1: R1<App>; declare const r2: R2<App>; declare const r3: R3<App>; declare const r6: R6<App>;
// structural: member type
const s1: R1<Wrong>["read"] = r1.read; // expect error
const s2: R2<Wrong>["read"] = r2.read; // expect error
const s3: R3<Wrong>["read"] = r3.read; // expect error
const s6: R6<Wrong>["read"] = r6.read; // expect error
// alias
const a6: R6<Wrong> = r6; // expect error
// usage still fine?
const ok3: string = r3.read(["theme"]).values.theme;
const ok6: string = r6.read(["theme", "region"]).values.region;
// slice
type Slice = { readonly theme: string };
const sl3: R3<Slice> = r3;
const sl6: R6<Slice> = r6;
const sls3: R3<Slice>["read"] = r3.read;
void [s1, s2, s3, s6, a6, ok3, ok6, sl3, sl6, sls3];
