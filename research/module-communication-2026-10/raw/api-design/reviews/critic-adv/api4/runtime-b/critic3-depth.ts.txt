// Critic round 3: cascade depth that grows with independent user actions through a leaked delivery context.
import { createServer, request, type Server } from "node:http";
import { createScope } from "@get-modular/resources";
import { createStore, type ListenerError } from "./observation.ts";
const reports: ListenerError[] = [];
const report = (r: ListenerError): undefined => { reports.push(r); return undefined; };
const host = createScope({ name: "host" });
const config = createStore<{ readonly listen: number }>({ listen: 0 }, { name: "config", resources: host.resources, reportError: report });
const stats = createStore<{ readonly hits: number }>({ hits: 0 }, { name: "stats", resources: host.resources, reportError: report });
let server: Server | undefined;
const start = (): void => {
  server?.close();
  server = createServer((req, res) => {           // every request is an independent user action
    try { stats.set({ hits: stats.read(["hits"]).values.hits + 1 }); res.end("ok"); } catch (error) { res.end(`failed: ${(error as { code?: string }).code}`); }
  });
  server.listen(0);
};
start();
const module = host.resources.child({ name: "m" });
const c = config.readonly(); const s = stats.readonly();
// Server module: restart when the config changes. Config owner: every 1 hit raises `listen` (a derived setting).
c.watch(c.read(["listen"]), { resources: module.resources, name: "server/restart", onChange: () => { start(); } });
s.watch(s.read(["hits"]), { resources: module.resources, name: "config/derive", onChange: ({ values }) => { config.set({ listen: values.hits }); } });
const call = (): Promise<string> => new Promise((resolve) => {
  const address = server?.address(); const port = typeof address === "object" && address !== null ? address.port : 0;
  request({ port, path: "/" }, (res) => { let body = ""; res.on("data", (x) => { body += String(x); }); res.on("end", () => { resolve(body); }); }).end();
});
const results: string[] = [];
for (let i = 0; i < 40; i += 1) { results.push(await call()); await new Promise((r) => { setTimeout(r, 5); }); }
const first = results.findIndex((r) => r !== "ok");
console.log(`depth: ${results.length} user requests; first failure at request ${first < 0 ? "none" : first + 1} (${first < 0 ? "" : results[first]}); reports ${[...new Set(reports.map((r) => (r.cause as { code?: string }).code))].join(",")}`);
console.log(`depth: cascade reports ${reports.length}, first by ${reports[0]?.subscription}; hits ${stats.read(["hits"]).values.hits}, derived listen ${config.read(["listen"]).values.listen}`);
server?.close();
