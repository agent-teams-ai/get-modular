import { createScope } from "scratchpad/api-shape/.toolchain/node_modules/@get-modular/resources/dist/index.js";
const [variant, file] = process.argv.slice(2);
const runtime = { a: "a-keys", b: "b-snapshot", c: "c-atomic" }[variant];
const { createStore } = await import(`./runtime/${runtime}.ts`);
const { createLoggerModule } = await import(`./${variant}/${file}.ts`);
const task = () => new Promise((r) => setImmediate(r));
const errors = [];
const host = createScope({ name: "host" });
const settings = createStore({ logLevel: "info", theme: "light" }, { name: "settings", resources: host.resources, reportError: (r) => { errors.push(r); } });
const mod = host.resources.child({ name: "logger" });
try {
  const pending = createLoggerModule({ settings: settings.readonly() }, { signal: mod.resources.signal, resources: mod.resources });
  await new Promise((r) => setTimeout(r, 2));
  settings.set({ logLevel: "error" });              // a change while the factory is still running
  const product = await pending;
  for (let i = 0; i < 5; i += 1) await task();
  const want = settings.read(["logLevel"]).values.logLevel;
  console.log(`[${variant}/${file}] store=${want} sink=${product.instance.threshold} => ${want === product.instance.threshold ? "OK" : "LOST UPDATE"}; reportError=${errors.length}`);
} catch (e) {
  console.log(`[${variant}/${file}] threw ${e.code ?? e.name}: ${e.message}`);
}
await mod.control.close(); await host.control.close();
