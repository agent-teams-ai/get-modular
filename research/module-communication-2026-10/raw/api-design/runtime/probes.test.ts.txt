// Lifetime, memory and scaling probes, run once per API variant against the real @get-modular/resources 0.1.0:
//   node --expose-gc runtime/probes.test.ts            (variants A and B; C was dropped after round 1)
//   VARIANT=b-snapshot node --expose-gc runtime/probes.test.ts
import assert from "node:assert/strict";
import { describe, test } from "node:test";
import { createScope, type CloseReport, type Resources } from "@get-modular/resources";
import { MAX_CASCADE_DEPTH, stats, type ListenerError, type RawSnapshot } from "./core.ts";
import { delivered, liveSubscriptions, settled } from "./testing.ts";
import { drivers, type Driver, type LoadHandlers, type LooseFacet, type LooseStore } from "./drivers.ts";

const gc = (globalThis as { readonly gc?: () => void }).gc;
const tick = (): Promise<void> => new Promise((resolve) => { setImmediate(resolve); });
const settle = delivered; // the testing entry point: no tuned tick counts
const collect = async (): Promise<void> => {
  assert.ok(gc, "run with node --expose-gc");
  for (let i = 0; i < 6; i += 1) { await tick(); gc(); }
};
const codeOf = (error: unknown): unknown => (error as { readonly code?: unknown } | null)?.code;
const debtsOf = (report: CloseReport): readonly { readonly path: string; readonly state: string; readonly code: unknown }[] =>
  report.debts.map((debt) => ({ path: debt.path.join("/"), state: debt.state, code: debt.state === "failed" ? codeOf(debt.cause) : undefined }));

type Deferred = { readonly promise: Promise<string>; readonly resolve: (value: string) => void };
function deferred(): Deferred {
  let resolve!: (value: string) => void;
  const promise = new Promise<string>((done) => { resolve = done; });
  return { promise, resolve };
}

type Limits = { readonly maxSubscriptionsPerScope?: number; readonly maxSubscriptions?: number };
function setup(driver: Driver, initial: Record<string, unknown> = { a: 0, b: 0, region: "r0" }, limits: Limits = {}) {
  const reports: ListenerError[] = [];
  const host = createScope({ name: "host" });
  const owner = host.resources.child({ name: "owner" });
  const store: LooseStore = driver.createStore(initial, {
    name: "settings", resources: owner.resources,
    reportError: (report) => { reports.push(report); return undefined; },
    ...limits,
  });
  const module = host.resources.child({ name: "module" });
  return { host, owner, module, store, facet: store.readonly(), reports };
}

/** A loader that ignores `signal` until released; records every call and every publication. */
function stubbornLoader() {
  const calls: { readonly region: unknown; readonly signal: AbortSignal; readonly gate: Deferred }[] = [];
  const published: unknown[] = [];
  const handlers: LoadHandlers = {
    loader: async (next, { signal }) => {
      const gate = deferred();
      calls.push({ region: next.values.region, signal, gate });
      return gate.promise;
    },
    onLoading: () => undefined,
    onResult: (result) => { published.push(result); return undefined; },
    onError: (cause) => { published.push(cause); return undefined; },
  };
  return { calls, published, handlers };
}

/** A loader that rejects as soon as its signal aborts. */
function cooperativeLoader() {
  const calls: AbortSignal[] = [];
  const published: unknown[] = [];
  const handlers: LoadHandlers = {
    loader: (_next, { signal }) => new Promise((_resolve, reject) => {
      calls.push(signal);
      signal.addEventListener("abort", () => { reject(signal.reason); }, { once: true });
    }),
    onLoading: () => undefined,
    onResult: (result) => { published.push(result); return undefined; },
    onError: (cause) => { published.push(cause); return undefined; },
  };
  return { calls, published, handlers };
}

const selected = process.env.VARIANT === undefined ? Object.values(drivers) : [drivers[process.env.VARIANT]];
for (const driver of selected) {
  if (driver === undefined) throw new Error(`unknown VARIANT ${process.env.VARIANT ?? ""}`);
  const follow = (facet: LooseFacet, resources: Resources, keys: readonly string[],
    name: string | undefined, onChange: (next: RawSnapshot) => undefined, between?: () => void) =>
    driver.follow(facet, resources, keys, name, onChange, between);

  describe(driver.variant, () => {
    test("1. a commit between reading and subscribing is not lost", async () => {
      const { store, facet, module } = setup(driver);
      let current: unknown;
      const first = follow(facet, module.resources, ["a"], "w", (next) => { current = next.values.a; return undefined; },
        () => { store.set({ a: 5 }); });
      current ??= first.values.a;
      await settle();
      assert.equal(current, 5);
    });

    test("2. the subscription is detached when its scope starts closing, before the drain finishes", async () => {
      const { store, facet, module } = setup(driver);
      const seen: unknown[] = [];
      follow(facet, module.resources, ["a"], "w", (next) => { seen.push(next.values.a); return undefined; });
      const load = stubbornLoader();
      driver.load(facet, module.resources, ["region"], "catalog", load.handlers);
      store.set({ a: 1 });
      await settle();
      assert.deepEqual(seen, [1]);
      assert.equal(load.calls.length, 1);
      assert.equal(stats(store).live, 2);

      let closed = false;
      const closing = module.control.close().then((report) => { closed = true; return report; });
      assert.equal(stats(store).live, 0, "detached synchronously by close()");
      assert.equal(load.calls[0]?.signal.aborted, true, "the running load is aborted at once");
      store.set({ a: 2, region: "r1" });
      await settle();
      assert.equal(closed, false, "the drain still waits for the stubborn load");
      assert.deepEqual(seen, [1], "no delivery after the scope started closing");
      assert.equal(load.calls.length, 1, "no new load after the scope started closing");
      load.calls[0]?.gate.resolve("late");
      const report = await closing;
      assert.deepEqual(debtsOf(report), []);
      assert.equal(report.complete, true);
      assert.deepEqual(load.published, [], "a result after close is dropped");
    });

    test("3. after the subscriber scope closes, the store holds no reference to its callbacks (GC)", async () => {
      const { host, store, facet } = setup(driver);
      let ref: WeakRef<object> | undefined;
      let closing: Promise<CloseReport> | undefined;
      (() => {
        const module = host.resources.child({ name: "module-gc" });
        const handler = (next: RawSnapshot): undefined => { void next; return undefined; };
        ref = new WeakRef(handler);
        follow(facet, module.resources, ["a"], "w", handler);
        closing = module.control.close();
      })();
      assert.equal((await closing)?.complete, true);
      await collect();
      assert.equal(ref?.deref(), undefined);
      assert.equal(stats(store).live, 0);
      assert.equal(stats(store).buckets, 0);
    });

    test("4. store.close() while the subscriber scope stays open: nothing is retained, one source.closed report each (GC, with control)", async () => {
      const retained = setup(driver);
      const released = setup(driver);
      const refs: WeakRef<object>[] = [];
      for (const { facet, module } of [retained, released]) {
        (() => {
          const handler = (next: RawSnapshot): undefined => { void next; return undefined; };
          refs.push(new WeakRef(handler));
          follow(facet, module.resources, ["a"], "w", handler); // module scopes stay open and referenced
        })();
      }
      released.store.close();
      assert.deepEqual(released.reports.map((r) => [r.source, r.subscription, codeOf(r.cause)]), [["settings", "w", "observation.source.closed"]]);
      assert.throws(() => released.facet.read(["a"]), (error: unknown) => codeOf(error) === "observation.store.closed");
      await collect();
      assert.notEqual(refs[0]?.deref(), undefined, "control: an open store keeps its live subscriber");
      assert.equal(refs[1]?.deref(), undefined, "a closed store's subscriber is released although its scope is open");
      assert.equal(liveSubscriptions(released.facet), 0);
      assert.throws(() => follow(released.facet, released.module.resources, ["a"], "w2", () => undefined),
        (error: unknown) => codeOf(error) === "observation.store.closed");
      assert.throws(() => released.store.set({ a: 1 }), (error: unknown) => codeOf(error) === "observation.store.closed");
      assert.equal((await released.module.control.close()).complete, true);
      void retained.module;
    });

    test("5. coalescing: 10 000 commits in one task give one delivery of the newest value", async () => {
      const { store, facet, module } = setup(driver);
      const seen: unknown[] = [];
      follow(facet, module.resources, ["a"], "w", (next) => { seen.push(next.values.a); return undefined; });
      let maxDirty = 0;
      for (let i = 1; i <= 10_000; i += 1) {
        store.set({ a: i });
        maxDirty = Math.max(maxDirty, stats(store).dirty);
      }
      await settle();
      assert.deepEqual(seen, [10_000]);
      assert.equal(maxDirty, 1, "one buffered entry per subscription, whatever the number of commits");
      assert.equal(stats(store).deliveries, 1);
    });

    test("6. load: one active and one pending run; a stale result never publishes", async () => {
      const { store, facet, module } = setup(driver);
      const load = stubbornLoader();
      driver.load(facet, module.resources, ["region"], "catalog", load.handlers);
      await settle();
      assert.equal(load.calls.length, 1);
      let maxPending = 0;
      for (let i = 1; i <= 100; i += 1) {
        store.set({ region: `r${i}` });
        await tick(); // one delivery per change: the pending slot is replaced 100 times
        maxPending = Math.max(maxPending, stats(store).loadsPending);
      }
      assert.equal(maxPending, 1);
      assert.equal(load.calls.length, 1, "no second run while the first is active");
      assert.equal(load.calls[0]?.signal.aborted, true, "the superseded run is aborted");
      load.calls[0]?.gate.resolve("result for r0");
      await settle();
      assert.equal(load.calls.length, 2, "only the newest change ran");
      assert.equal(load.calls[1]?.region, "r100");
      load.calls[1]?.gate.resolve("result for r100");
      await settle();
      assert.deepEqual(load.published, ["result for r100"], "every publication is recorded: the stale one never appeared");
      await module.control.close();
    });

    test("7. dispatch visits only subscriptions of the changed keys (hundreds of watchers)", async (t) => {
      for (const watchers of [10, 100, 1000]) {
        const { store, facet, module } = setup(driver, { a: 0, b: 0, region: "r0" }, { maxSubscriptionsPerScope: 2048 });
        for (let i = 0; i < watchers; i += 1) follow(facet, module.resources, ["a"], `a${i}`, () => undefined);
        follow(facet, module.resources, ["b"], "b", () => undefined);
        const before = stats(store).visits;
        store.set({ b: 1 });
        assert.equal(stats(store).visits - before, 1, `${watchers} watchers on a: a change of b visits one`);
        store.set({ a: 1 });
        assert.equal(stats(store).visits - before - 1, watchers);
        const started = performance.now();
        for (let i = 2; i < 2_002; i += 1) store.set({ b: i });
        t.diagnostic(`${driver.variant}: ${watchers} watchers on a, set(b) x2000: ${((performance.now() - started) / 2).toFixed(3)} us per set`);
        await settle();
        await module.control.close();
      }
    });

    test("8. a repeated watch in a long-lived scope fails with a coded error, in that scope only", async () => {
      const { facet, module, store, host } = setup(driver);
      follow(facet, module.resources, ["a"], "per-request", () => undefined);
      assert.throws(() => follow(facet, module.resources, ["a"], "per-request", () => undefined),
        (error: unknown) => codeOf(error) === "observation.subscription.duplicate-name");
      assert.equal(stats(store).live, 1);
      // Generated names bypass the name check; the per-scope limit (default 64) stops them in the leaking scope.
      let accepted = 1;
      assert.throws(() => {
        for (let i = 0; i < 1000; i += 1) { follow(facet, module.resources, ["a"], `request-${i}`, () => undefined); accepted += 1; }
      }, (error: unknown) => codeOf(error) === "observation.subscriptions.limit");

      // Another module is not affected by the leak.
      const other = host.resources.child({ name: "other" });
      follow(facet, other.resources, ["a"], "other/a", () => undefined);
      assert.equal(liveSubscriptions(facet, other.resources), 1);
      assert.equal(liveSubscriptions(facet, module.resources), 64);
      await module.control.close();
      await other.control.close();
      assert.equal(stats(store).live, 0);
    });

    test("10. limits: 64 per subscriber scope by default; the owner's per-store safety limit across scopes", async () => {

    test("11. closing the subscriber scope during a cooperative load: complete, no debts, nothing published", async () => {
      const { facet, module } = setup(driver);
      const load = cooperativeLoader();
      driver.load(facet, module.resources, ["region"], "catalog", load.handlers);

      const report = await module.control.close();
      assert.equal(report.complete, true);
      assert.deepEqual(debtsOf(report), []);
      assert.equal(load.calls[0]?.aborted, true);
      assert.deepEqual(load.published, [], "an aborted run reports nothing");
    });

    test("12. a load that ignores abort: escalate gives a failed debt with a code; a retry completes", async () => {
      const { facet, module } = setup(driver);
      const load = stubbornLoader();
      driver.load(facet, module.resources, ["region"], "catalog", load.handlers);
      await settle();
      const escalate = new AbortController();
      const closing = module.control.close({ escalate: escalate.signal });
      escalate.abort();
      const report = await closing;
      assert.equal(report.complete, false);
      assert.deepEqual(debtsOf(report), [
        { path: "host/module/load:catalog/drain", state: "failed", code: "observation.load.still-running" },
      ]);
      load.calls[0]?.gate.resolve("late");
      await settle();
      assert.deepEqual(load.published, [], "a late result after close is dropped");
      const retry = await module.control.close();
      assert.equal(retry.complete, true);
    });

    test("13. a load that ignores abort: abandon gives a pending debt and settled: false", async () => {
      const { facet, module } = setup(driver);
      const load = stubbornLoader();
      driver.load(facet, module.resources, ["region"], "catalog", load.handlers);
      await settle();
      const abandon = new AbortController();
      const closing = module.control.close({ abandon: abandon.signal });
      await settle();
      abandon.abort();
      const report = await closing;
      assert.equal(report.settled, false);
      assert.deepEqual(debtsOf(report), [{ path: "host/module/load:catalog/drain", state: "pending", code: undefined }]);
      load.calls[0]?.gate.resolve("late");
      const final = await module.control.close();
      assert.equal(final.complete, true);
    });

    test("14. store.close() during a load aborts it; the drain stays in the subscriber scope", async () => {
      const { store, facet, module, owner, reports } = setup(driver);
      const load = stubbornLoader();
      driver.load(facet, module.resources, ["region"], "catalog", load.handlers);
      await settle();
      store.close();
      assert.equal(load.calls[0]?.signal.aborted, true);
      assert.equal(stats(store).live, 0);
      assert.deepEqual(reports.map((r) => codeOf(r.cause)), ["observation.source.closed"]);
      load.calls[0]?.gate.resolve("late");
      await settle();
      assert.deepEqual(load.published, []);
      const moduleReport = await module.control.close();
      assert.equal(moduleReport.complete, true);
      const ownerReport = await owner.control.close();
      assert.equal(ownerReport.complete, true, "store close is idempotent at the owner's LIFO slot");
    });

    test("17. load: onLoading when a newer run is queued; a result that lands after set() is dropped", async () => {
      const { store, facet, module } = setup(driver);
      const events: string[] = [];
      const gates: { readonly region: unknown; readonly release: (value: string) => void }[] = [];
      driver.load(facet, module.resources, ["region"], "catalog", {
        loader: (next) => new Promise<string>((resolve) => { gates.push({ region: next.values.region, release: resolve }); }),
        onLoading: (snapshot) => { events.push(`loading ${String(snapshot.values.region)}`); return undefined; },
        onResult: (result) => { events.push(`result ${String(result)}`); return undefined; },
        onError: (cause) => { events.push(`error ${String(cause)}`); return undefined; },
      });
      assert.deepEqual(events, [], "nothing is called back synchronously");
      await settle();
      assert.deepEqual(events, ["loading r0"], "the first run is announced");
      gates[0]?.release("r0 models");
      await settle();
      store.set({ region: "jp" });
      await settle();
      assert.deepEqual(events.slice(1), ["result r0 models", "loading jp"]);
      store.set({ region: "br" });                // br chosen; jp still running and not yet aborted
      gates[1]?.release("jp models");             // jp lands in the same task, before the delivery of br
      await settle();
      assert.deepEqual(events.slice(3), ["loading br"], "the jp result is dropped against the store revision");
      gates[2]?.release("br models");
      await settle();
      assert.deepEqual(events.slice(4), ["result br models"]);
      await module.control.close();
    });

    test("18. echo: a handler that changes a key it follows fails at the first delivery, reported and thrown", async () => {
      const { store, facet, module, reports } = setup(driver);
      let calls = 0;
      follow(facet, module.resources, ["a"], "echo", (next) => {
        calls += 1;
        if (calls <= 1000) store.set({ a: Number(next.values.a) + 1 }); // the probe's own stop if the guard is missing
        return undefined;
      });
      store.set({ a: 1 });
      await settle();
      assert.equal(calls, 1, "stopped at the first round, not after 64");
      assert.equal(store.read(["a"]).values.a, 1, "the echoing commit changed nothing");
      assert.deepEqual(reports.map((r) => [r.subscription, codeOf(r.cause)]), [["echo", "observation.delivery.echo"]], "reported once");
      // Writing another key of the same store, or another store, is not an echo.
      follow(facet, module.resources, ["region"], "derive-b", (next) => { store.set({ b: String(next.values.region).length }); return undefined; });
      store.set({ region: "abcd" });
      await settle();
      assert.equal(store.read(["b"]).values.b, 4);

    });

    test("19. cascade guard: shared across stores (ping-pong), legitimate chains pass, coalescing keeps the largest depth", async () => {
      const first = setup(driver);
      const second = setup(driver);
      let calls = 0;
      follow(first.facet, first.module.resources, ["a"], "ping", (next) => { calls += 1; if (calls <= 1000) second.store.set({ a: Number(next.values.a) + 1 }); return undefined; });
      follow(second.facet, second.module.resources, ["a"], "pong", (next) => { calls += 1; if (calls <= 1000) first.store.set({ a: Number(next.values.a) + 1 }); return undefined; });
      first.store.set({ a: 1 });
      await settle();
      assert.equal(calls, MAX_CASCADE_DEPTH + 1, "one bound for the whole chain across both stores");
      assert.equal(first.reports.length + second.reports.length, 1);
      // A chain of 20 derived stores, each set by a handler of the previous one, completes.
      const chain = Array.from({ length: 21 }, () => setup(driver));
      chain.slice(0, 20).forEach((link, i) => {
        follow(link.facet, link.module.resources, ["a"], `link-${i}`, (next) => { chain[i + 1]?.store.set({ a: next.values.a }); return undefined; });
      });
      chain[0]?.store.set({ a: 42 });
      await settle();
      assert.equal(chain[20]?.store.read(["a"]).values.a, 42);
      assert.equal(chain.reduce((n, link) => n + link.reports.length, 0), 0);
      // 10 000 commits at depth 0 coalesce into one delivery at depth 0, not at depth 10 000.
      const burst = setup(driver);
      let echoes = 0;
      follow(burst.facet, burst.module.resources, ["a"], "burst", (next) => { echoes += 1; if (echoes === 1) burst.store.set({ b: Number(next.values.a) }); return undefined; });
      for (let i = 1; i <= 10_000; i += 1) burst.store.set({ a: i });
      await settle();
      assert.equal(burst.reports.length, 0);
      assert.equal(burst.store.read(["b"]).values.b, 10_000);
    });

    test("22. asynchronous echo is caught at the first round: async command, microtask, loader body (Node)", async () => {
      const { store, facet, module, reports, host } = setup(driver);
      const thrown: unknown[] = [];
      // C1: an async command called from a handler

      let c1 = 0;

    });

    test("23. an asynchronous ping-pong across two stores stops at 64 steps (Node)", async () => {
      const first = setup(driver);
      const second = setup(driver);
      let calls = 0;
      const thrown: unknown[] = [];
      const command = (target: LooseStore) => async (a: number): Promise<void> => { await Promise.resolve(); target.set({ a }); };
      const toSecond = command(second.store);
      const toFirst = command(first.store);
      follow(first.facet, first.module.resources, ["a"], "ping", (next) => { calls += 1; if (calls <= 1000) toSecond(Number(next.values.a) + 1).catch((e: unknown) => { thrown.push(e); }); return undefined; });
      follow(second.facet, second.module.resources, ["a"], "pong", (next) => { calls += 1; if (calls <= 1000) toFirst(Number(next.values.a) + 1).catch((e: unknown) => { thrown.push(e); }); return undefined; });
      first.store.set({ a: 1 });
      for (let round = 0; round < 3; round += 1) await settle();
      assert.equal(calls, MAX_CASCADE_DEPTH + 1);
      assert.deepEqual(thrown.map(codeOf), ["observation.delivery.cascade"]);
      assert.equal(first.reports.length + second.reports.length, 1);
    });

    test("24. pagination as a loop inside one loader passes 100 pages", async () => {
      const { facet, module } = setup(driver);
      let rows: unknown[] = [];
      driver.load(facet, module.resources, ["region"], "pages", {
        loader: async () => { const all: number[] = []; for (let page = 0; page < 100; page += 1) { await Promise.resolve(); all.push(page); } return all; },
        onLoading: () => undefined, onResult: (result) => { rows = result as unknown[]; return undefined; }, onError: () => undefined,
      });
      await settled();
      assert.equal(rows.length, 100);
      await module.control.close();
    });

    test("25. the owner's read works after close; a save-on-close entry registered after createStore sees the last values", async () => {
      const host = createScope({ name: "host" });
      const module = host.resources.child({ name: "prefs" });
      const saved: unknown[] = [];
      const store = driver.createStore({ theme: "light" }, { name: "prefs", resources: module.resources, reportError: () => undefined });
      module.resources.use({ [Symbol.dispose]: () => { saved.push(store.read(["theme"]).values.theme); } }, "prefs/save");
      store.set({ theme: "dark" });
      const report = await module.control.close();
      assert.equal(report.complete, true);
      assert.deepEqual(saved, ["dark"]);
      assert.equal(store.read(["theme"]).values.theme, "dark", "the owner still reads after close");
      assert.throws(() => store.readonly().read(["theme"]), (error: unknown) => codeOf(error) === "observation.store.closed");
    });

    test("26. a failed set leaves the caller's objects unfrozen; the global limit names scope counts", () => {
      const { store } = setup(driver, { a: 0, b: 0, region: "r0", rows: [] });
      const mine = { list: [1, 2] };
      assert.throws(() => store.set({ a: mine, b: new Date() }), (error: unknown) => codeOf(error) === "observation.values.not-plain");
      assert.equal(Object.isFrozen(mine), false);
      assert.equal(Object.isFrozen(mine.list), false);
      const limited = setup(driver, { a: 0, b: 0, region: "r0" }, { maxSubscriptions: 3 });
      for (let i = 0; i < 3; i += 1) follow(limited.facet, limited.host.resources.child({ name: `request-${i}` }).resources, ["a"], "m/a", () => undefined);
      assert.throws(() => follow(limited.facet, limited.module.resources, ["a"], "innocent/a", () => undefined),
        (error: unknown) => codeOf(error) === "observation.subscriptions.limit" && /in 3 scopes, at most 1 in one/.test((error as Error).message));
    });

    test("15. callback failures go to reportError with the subscription name", async () => {
      const { store, facet, module, reports } = setup(driver);
      follow(facet, module.resources, ["a"], "throws", () => { throw new Error("boom"); });
      follow(facet, module.resources, ["a"], "returns", ((): unknown => 1) as () => undefined); // a JS caller
      store.set({ a: 1 });
      await settle();
      assert.deepEqual(reports.map((r) => [r.subscription, codeOf(r.cause) ?? (r.cause as Error).message]),
        [["throws", "boom"], ["returns", "observation.callback.returned-value"]]);
      assert.ok(reports.every((r) => r.source === "settings"));
      await module.control.close();
    });

    test("16. after subscribing, nothing is called back synchronously", () => {

      follow(facet, module.resources, ["a"], "w", () => { calls += 1; return undefined; }, () => { store.set({ a: 9 }); });
      assert.equal(calls, 0);
    });
