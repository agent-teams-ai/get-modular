import type { ModuleContext } from "@get-modular/resources";
import type { Screen, Session, SessionsDeps, SessionsProduct } from "./e-sessions.types.js";

export async function createSessionsModule(deps: SessionsDeps, { resources }: ModuleContext): Promise<SessionsProduct> {
  const open = (user: string, screen: Screen): Session => {
    const session = resources.child({ name: `session:${user}` }); // the subscription ends with the session
    const banner = deps.settings.watch({
      resources: session.resources, name: "banner", keys: ["banner"],
      onChange: ({ values }) => { screen.show(values.banner); },
    });
    screen.show(banner.values.banner);
    return {
      end: async () => {
        const report = await session.control.close();
        if (!report.complete) throw new Error(`session ${user} did not release everything`);
      },
    };
  };
  return { instance: undefined, capabilities: { "acme/sessions": { open } } };
}
