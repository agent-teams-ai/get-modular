import type { ModuleContext } from "@get-modular/resources";
import type { QuotesDeps, QuotesProduct, Screen } from "./h-requests.types.js";

export async function createQuotesModule(deps: QuotesDeps, { resources }: ModuleContext): Promise<QuotesProduct> {
  // One subscription for all tickers: no scope per ticker, so nothing is registered per open ticker.
  const screens = new Set<Screen>();
  deps.settings.watch(deps.settings.read(["eurUsd"]), {
    resources,
    name: "quotes/rate",
    onChange: ({ values }) => {
      const failures: unknown[] = [];
      for (const screen of screens) {
        try {
          screen.show(values.eurUsd);
        } catch (error) {
          failures.push(error);
        }
      }
      if (failures.length > 0) {
        throw new AggregateError(failures, "ticker screens failed to show the rate");
      }
    },
  });
  return {
    instance: undefined,
    capabilities: {
      "acme/quotes": {
        convert: (eur) => eur * deps.settings.read(["eurUsd"]).values.eurUsd,
        openTicker: (screen) => {
          screen.show(deps.settings.read(["eurUsd"]).values.eurUsd);
          screens.add(screen);
          return {
            close: async () => {
              screens.delete(screen);
            },
          };
        },
      },
    },
  };
}
