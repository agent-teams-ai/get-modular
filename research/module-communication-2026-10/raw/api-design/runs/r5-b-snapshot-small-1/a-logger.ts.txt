import type { ModuleContext } from "@get-modular/resources";
import { createConsoleSink, type LoggerDeps, type LoggerProduct } from "./a-logger.types.js";

export async function createLoggerModule(deps: LoggerDeps, { resources }: ModuleContext): Promise<LoggerProduct> {
  const logLevel = deps.settings.read(["logLevel"]);
  const sink = createConsoleSink(logLevel.values.logLevel);
  deps.settings.watch(logLevel, {
    resources,
    name: "logger/log-level",
    onChange: ({ values }) => {
      sink.setThreshold(values.logLevel);
    },
  });
  return {
    instance: sink,
    capabilities: {
      "acme/logger": {
        log: (level, message) => {
          sink.write(level, message);
        },
      },
    },
  };
}
