import type { ModuleContext } from "@get-modular/resources";
import { createGreeter, type GreeterDeps, type GreeterProduct } from "./d-greeter.types.js";

export async function createGreeterModule(deps: GreeterDeps, _context: ModuleContext): Promise<GreeterProduct> {
  // Read once: a later locale takes effect only when the module is created again.
  const greeter = createGreeter(deps.settings.read(["locale"]).values.locale);
  return { instance: undefined, capabilities: { "acme/greeter": greeter } };
}
