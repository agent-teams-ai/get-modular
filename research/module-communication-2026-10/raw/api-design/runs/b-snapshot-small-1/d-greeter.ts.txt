import type { ModuleContext } from "@get-modular/resources";
import { createGreeter, type GreeterDeps, type GreeterProduct } from "./d-greeter.types.js";

export async function createGreeterModule(deps: GreeterDeps, context: ModuleContext): Promise<GreeterProduct> {
  const { values } = deps.settings.read(["locale"]);
  const greeter = createGreeter(values.locale);
  return {
    instance: undefined,
    capabilities: {
      "acme/greeter": greeter,
    },
  };
}
