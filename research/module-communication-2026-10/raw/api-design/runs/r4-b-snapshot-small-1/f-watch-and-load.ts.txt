import type { ModuleContext } from "@get-modular/resources";
import {
  fetchHolidays, regionLabel, type Holidays, type PanelDeps, type PanelProduct, type PanelView,
} from "./f-watch-and-load.types.js";

export async function createRegionPanelModule(deps: PanelDeps, { resources }: ModuleContext): Promise<PanelProduct> {
  const region = deps.settings.read(["region"]);
  let label = regionLabel(region.values.region);
  let holidays: Holidays = { status: "loading" };
  deps.settings.watch(region, {
    resources,
    name: "region-panel/label",
    onChange: ({ values }) => {
      label = regionLabel(values.region);
    },
  });
  deps.settings.load(region, {
    resources,
    name: "region-panel/holidays",
    loader: ({ values }, { signal }) => fetchHolidays(values.region, { signal }),
    onLoading: () => {
      holidays = { status: "loading" };
    },
    onResult: (days, { values }) => {
      holidays = { status: "ready", region: values.region, days };
    },
    onError: (cause, { values }) => {
      holidays = { status: "failed", region: values.region, cause };
    },
  });
  const currentHolidays = (): Holidays => {
    // A newer region may be chosen but not yet delivered to this module: never show an older region's holidays.
    if (holidays.status !== "loading" && holidays.region !== deps.settings.read(["region"]).values.region) {
      return { status: "loading" };
    }
    return holidays;
  };
  const view = (): PanelView => ({ label, holidays: currentHolidays() });
  return { instance: undefined, capabilities: { "acme/region-panel": { view } } };
}
