import type { ModuleContext } from "@get-modular/resources";
import { createStore } from "@get-modular/observation";
import {
  createNotifier, parseQuietHours, type NotificationStatus, type NotificationsDeps, type NotificationsProduct,
} from "./i-errors.types.js";

export async function createNotificationsModule(deps: NotificationsDeps, { resources }: ModuleContext): Promise<NotificationsProduct> {
  const settings = deps.settings.read(["quietHours"]);
  const notifier = createNotifier(parseQuietHours(settings.values.quietHours));
  deps.settings.watch(settings, {
    resources,
    name: "notifications/quiet-hours",
    onChange: ({ revision, values }) => {
      try {
        notifier.setQuietHours(parseQuietHours(values.quietHours));
      } catch (cause) {
        deps.errors.report({ store: "settings", subscription: "notifications/quiet-hours", revision, cause });
      }
    },
  });
  const status = createStore<NotificationStatus>({ muted: false }, {
    name: "notifications/status",
    resources,
    reportError: deps.errors.report,
  });
  return {
    instance: notifier,
    capabilities: {
      "acme/notifications": {
        mute: (muted) => {
          status.set({ muted });
        },
      },
      "acme/notifications/status": status.readonly(),
    },
  };
}
