import type { ModuleContext } from "@get-modular/resources";
import { openAuditLog, type AuditDeps, type AuditProduct } from "./e-async-factory.types.js";

export async function createAuditModule(deps: AuditDeps, { resources }: ModuleContext): Promise<AuditProduct> {
  const retention = deps.settings.read(["retentionDays"]);
  const log = await resources.setup({
    name: "audit/log",
    setup: ({ signal }) => openAuditLog(retention.values.retentionDays, { signal }),
    cleanup: (opened) => opened.close(),
  });
  deps.settings.watch(retention, {
    resources, name: "audit/retention",
    onChange: ({ values }) => { log.setRetention(values.retentionDays); },
  });
  return {
    instance: log,
    capabilities: { "acme/audit": { record: (entry): undefined => { log.append(entry); return undefined; } } },
  };
}
