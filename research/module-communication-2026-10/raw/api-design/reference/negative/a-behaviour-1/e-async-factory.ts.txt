// Wrong on purpose (variant A): changesAfter is a snapshot read after the await, so a change during the await is lost.
import type { ModuleContext } from "@get-modular/resources";
import { openAuditLog, type AuditDeps, type AuditProduct } from "./e-async-factory.types.js";

export async function createAuditModule(deps: AuditDeps, { resources }: ModuleContext): Promise<AuditProduct> {
  const log = await resources.setup({
    name: "audit/log",
    setup: ({ signal }) => openAuditLog(deps.settings.read(["retentionDays"]).values.retentionDays, { signal }),
    cleanup: (opened) => opened.close(),
  });
  deps.settings.watch(resources, {
    name: "audit/retention", keys: ["retentionDays"], changesAfter: deps.settings.read(["retentionDays"]),
    onChange: ({ values }) => { log.setRetention(values.retentionDays); },
  });
  return { instance: log, capabilities: { "acme/audit": { record: (entry) => { log.append(entry); return undefined; } } } };
}
