// S0 0A: positive and negative type cases on tsc 7.0.2 and TS 5.8.3, NodeNext and Bundler.
//   node negatives.mjs --workspace <built prototype checkout>
// Every file in negatives/ except positive.ts and shared.ts states `// expect: ...` on its first line; a file whose
// expectation starts with NO error must compile; every other file must fail with the TS code its expectation names first
// (positions are not checked), the same code in all four runs.
import { spawnSync } from "node:child_process";
import { cp, mkdir, mkdtemp, readFile, readdir, realpath, rm, symlink, writeFile } from "node:fs/promises";
import { createRequire } from "node:module";
import { tmpdir } from "node:os";
import { dirname, join, resolve } from "node:path";
import { parseArgs } from "node:util";
import { fileURLToPath } from "node:url";

const { values } = parseArgs({ options: { workspace: { type: "string" } } });
if (!values.workspace) throw new Error("--workspace is required");
const workspace = resolve(values.workspace);
const here = dirname(fileURLToPath(import.meta.url));
const require = createRequire(join(workspace, "package.json"));

const directory = await realpath(await mkdtemp(join(tmpdir(), "gm-s0-negatives-")));
try {
  await mkdir(join(directory, "node_modules/@get-modular"), { recursive: true });
  for (const name of ["core", "assembly", "resources", "conformance"]) {
    await symlink(join(workspace, "packages", name), join(directory, "node_modules/@get-modular", name), "junction");
  }
  await writeFile(join(directory, "package.json"), JSON.stringify({ name: "negatives", private: true, type: "module" }));
  const files = (await readdir(join(here, "negatives"))).filter((name) => name.endsWith(".ts")).sort();
  for (const name of files) await cp(join(here, "negatives", name), join(directory, name));
  await writeFile(join(directory, "tsconfig.json"), JSON.stringify({ compilerOptions: {
    target: "ES2022", lib: ["ES2023", "ESNext.Disposable", "DOM"], strict: true, noEmit: true,
    skipLibCheck: false, types: [], isolatedDeclarations: false, erasableSyntaxOnly: false,
  }, files }));
  const expectations = new Map();
  for (const name of files) {
    const first = (await readFile(join(here, "negatives", name), "utf8")).split("\n", 1)[0];
    expectations.set(name, /^\/\/ expect: (.*)$/u.exec(first)?.[1]);
  }
  const results = new Map();
  for (const compilerName of ["typescript", "typescript-minimum"]) {
    const manifestPath = require.resolve(`${compilerName}/package.json`);
    const manifest = JSON.parse(await readFile(manifestPath, "utf8"));
    const compiler = join(dirname(manifestPath), "bin/tsc");
    for (const [module, resolution] of [["NodeNext", "NodeNext"], ["ESNext", "Bundler"]]) {
      const run = spawnSync(process.execPath, [compiler, "--project", join(directory, "tsconfig.json"), "--module", module,
        "--moduleResolution", resolution, "--pretty", "false"], { cwd: directory, encoding: "utf8", maxBuffer: 16 * 1024 * 1024,
        env: { ...process.env, NODE_PATH: "", NODE_OPTIONS: "" } });
      const label = `${manifest.version} ${resolution}`;
      const byFile = new Map();
      for (const m of `${run.stdout}\n${run.stderr}`.matchAll(/^(\S+?\.ts)\((\d+),\d+\): error (TS\d+): (.*)$/gmu)) {
        const list = byFile.get(m[1]) ?? [];
        list.push({ line: Number(m[2]), code: m[3], message: m[4] });
        byFile.set(m[1], list);
      }
      results.set(label, byFile);
    }
  }
  let failures = 0;
  const rows = [];
  for (const name of files) {
    const expected = expectations.get(name);
    const codes = [...results].map(([label, byFile]) => [label, [...new Set((byFile.get(name) ?? []).map((d) => d.code))].sort().join("+") || "none"]);
    const unique = new Set(codes.map(([, c]) => c));
    const mustPass = name === "positive.ts" || name === "shared.ts" || expected?.startsWith("NO error");
    const wanted = /^(TS\d+)/u.exec(expected ?? "")?.[1];
    const ok = mustPass ? unique.size === 1 && unique.has("none") : unique.size === 1 && !unique.has("none") && (wanted === undefined || unique.has(wanted));
    if (!ok) failures += 1;
    const first = (results.values().next().value.get(name) ?? [])[0];
    rows.push({ file: name, ok, codes: Object.fromEntries(codes), expected, firstMessage: first?.message?.slice(0, 200) });
    console.error(`${ok ? "ok  " : "FAIL"} ${name}: ${[...unique].join(" | ")}`);
  }
  console.log(JSON.stringify({ rows }, null, 2));
  process.exitCode = failures === 0 ? 0 : 1;
} finally {
  await rm(directory, { recursive: true, force: true });
}
