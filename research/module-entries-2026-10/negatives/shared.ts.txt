// Shared fixture of the type cases: two features that share one contract, a Host map over both.
import { optional, required } from "@get-modular/core";
import { declareModule, defineContract, group, implement } from "@get-modular/assembly";
import type { CapabilitiesOf } from "@get-modular/assembly";
import { scoped } from "@get-modular/resources";

export const Clock = defineContract<{ readonly now: () => number }>()({ id: "s0/clock", revision: 1 });
export const Orders = defineContract<{ readonly count: () => number }>()({ id: "s0/orders", revision: 1 });
export const Audit = defineContract<{ readonly note: (text: string) => void }>()({ id: "s0/audit", revision: 1 });
export const ClockV2 = defineContract<{ readonly now: () => number }>()({ id: "s0/clock", revision: 2 });
const owner = { authority: "s0", path: ["types"] };

export interface TimeMap extends CapabilitiesOf<typeof Clock> {}
export interface OrdersMap extends CapabilitiesOf<typeof Clock | typeof Orders | typeof Audit> {}
export interface HostMap extends CapabilitiesOf<typeof Clock | typeof Orders | typeof Audit> {}

export const clockDeclaration = declareModule({
  moduleId: "s0/clock", implementationId: "s0/clock/system", owner, provides: [Clock.provide()], slots: [],
});
export const ordersDeclaration = declareModule({
  moduleId: "s0/orders", implementationId: "s0/orders/default", owner, provides: [Orders.provide()],
  slots: [Clock.slot("clock", required()), Audit.slot("audit", optional())],
});

export const clockEntry = implement<TimeMap>()(clockDeclaration, async () => ({
  instance: { kind: "clock" as const }, capabilities: { "s0/clock": { now: () => 1 } },
}));
export const ordersEntry = implement<OrdersMap>()(ordersDeclaration, async (deps, context) => ({
  instance: { total: deps.clock.now() + (context.resources === undefined ? 0 : 1) }, capabilities: { "s0/orders": { count: () => 2 } },
}), scoped);
export const timeGroup = group("time", [clockEntry]);
export const ordersGroup = group("orders", [ordersEntry]);
