// Internal: one isolation rule for every user callback. Throws and illegal
// return values (including Promises smuggled through casts) go to the reporter;
// the caller's loop always continues. Reporter contract: synchronous, no-throw.
// A throwing reporter is a Host bug: it is rethrown in a later microtask so it
// surfaces as an uncaught error (deliberately loud, like RxJS), never swallowed.

import { ObservationError } from "./contract.ts";

export type Report = (cause: unknown) => undefined;

const isThenable = (value: unknown): value is PromiseLike<unknown> =>
  (typeof value === "object" || typeof value === "function") && value !== null &&
  typeof (value as { then?: unknown }).then === "function";

export function safeReport(report: Report, cause: unknown): void {
  try {
    report(cause);
  } catch (reporterFailure) {
    queueMicrotask(() => { throw new AggregateError([cause, reporterFailure], "observation reportError threw"); });
  }
}

export function invokeIsolated(call: () => unknown, report: Report): void {
  let result: unknown;
  try {
    result = call();
  } catch (cause) {
    safeReport(report, cause);
    return;
  }
  if (result === undefined) return;
  safeReport(report, new ObservationError("observation.callback.returned-value", "callbacks must be synchronous and return undefined"));
  try {
    if (isThenable(result)) result.then(undefined, (cause: unknown) => safeReport(report, cause));
  } catch (cause) {
    safeReport(report, cause); // a hostile `then` must not escape the caller's loop
  }
}
