// Feature composition (module factory). OPTION 1 = sync participant:
// one construction read, then PROVIDE `syncSettings(initial, apply)`.

import { syncSettings, type SettingsReaderV1 } from "../settings/index.ts";
import { Greeter } from "./application/greeter.ts";

export type GreeterApi = Readonly<{ greet: (name: string) => string }>;

export async function createGreeting(deps: Readonly<{ settings: SettingsReaderV1 }>) {
  const initial = deps.settings.read(["locale", "greetingStyle"]);
  const greeter = new Greeter({ locale: initial.values.locale, style: initial.values.greetingStyle });
  return {
    instance: greeter,
    capabilities: {
      "greeting/greeter": { greet: (name: string) => greeter.greet(name) } satisfies GreeterApi,
      "settings/participant": syncSettings(initial, ({ values }) =>
        greeter.configure({ locale: values.locale, style: values.greetingStyle })),
    },
  };
}
