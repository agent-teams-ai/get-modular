// PASSIVE modules: what each of the hundreds of non-participants looks like.
// `audit-log` reads one setting at construction (IOptions-style) and keeps it.
// `clock` has no settings dependency. Neither knows that hooks exist.

import type { SettingsReaderV1 } from "../settings/index.ts";

export type AuditApi = Readonly<{ record: (event: string) => string }>;

export async function createAuditLog(deps: Readonly<{ settings: SettingsReaderV1 }>) {
  const verbose = deps.settings.read(["auditVerbose"]).values.auditVerbose;
  const audit: AuditApi = { record: (event) => (verbose ? `[verbose] ${event}` : event) };
  return { instance: { verbose }, capabilities: { "audit/log": audit } };
}

export type ClockApi = Readonly<{ now: () => number }>;

export async function createClock() {
  let ticks = 0;
  const clock: ClockApi = { now: () => ++ticks };
  return { instance: clock, capabilities: { "time/clock": clock } };
}
