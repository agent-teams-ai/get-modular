// Feature composition. OPTION 3 = derived participant: PROVIDE pure `derive` +
// synchronous `commit`/`fail`. No subscription, lane, cleanup or readiness code.

import { derivedSettings } from "../settings/index.ts";
import { TaxTable, type TaxRulesPort } from "./application/tax-table.ts";

export type { TaxRulesPort } from "./application/tax-table.ts";
export type TaxApi = Readonly<{ vat: () => number | undefined }>;

export async function createTaxTable(deps: Readonly<{ rules: TaxRulesPort }>) {
  const table = new TaxTable();
  return {
    instance: table,
    capabilities: {
      "tax/table": { vat: () => table.vat() } satisfies TaxApi,
      "settings/participant": derivedSettings({
        keys: ["taxRegion"],
        derive: ({ taxRegion }, signal) => deps.rules.load(taxRegion, signal),
        commit: (rules) => table.replace(rules),
        fail: (cause, { revision }) => table.recordFailure(revision, cause),
      }),
    },
  };
}
